  target_link_libraries(ista_sdc_frontend INTERFACE ista_sdc_writer)
endif()

  target_link_libraries(ista_data_arrival_propagator INTERFACE ista_data_slew_propagator)
elseif(CASE STREQUAL "sdc_core_frontend")
  target_link_libraries(ista_sdc INTERFACE ista_sdc_frontend)
elseif(CASE STREQUAL "sdc_core_writer")

  target_link_libraries(ista_sdc_query INTERFACE ista_sdc_frontend)
elseif(CASE STREQUAL "catalog_query")
  target_link_libraries(ista_sdc_catalog INTERFACE ista_sdc_query)
elseif(CASE STREQUAL "sdf_analysis")
  target_link_libraries(ista_sdf_writer INTERFACE ista_timing_analyzer)

  target_link_libraries(ista_sdc_writer INTERFACE ista_data_manager)
